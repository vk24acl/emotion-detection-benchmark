    # Step 4: Custom PyTorch Dataset class
    class EmotionDataset(Dataset):
        def __init__(self, encodings, labels):
            self.encodings = encodings
            self.labels    = labels

        def __getitem__(self, idx):
            item = {key: torch.tensor(val[idx]) for key, val in self.encodings.items()}
            item['labels'] = torch.tensor(self.labels[idx])
            return item

        def __len__(self):
            return len(self.labels)

    # Step 5: Shared helper functions

    def tokenize_texts(texts, tokenizer):
        return tokenizer(
            list(texts),
            padding='max_length',
            truncation=True,
            max_length=MAX_LENGTH,
            return_tensors=None
        )

    def compute_metrics(eval_pred):
        predictions, labels = eval_pred
        predictions = np.argmax(predictions, axis=-1)
        return {'accuracy': accuracy_score(labels, predictions)}

    def get_training_args(output_dir, epochs=NUM_EPOCHS):
        return TrainingArguments(
            output_dir=output_dir,
            num_train_epochs=epochs,
            per_device_train_batch_size=BATCH_SIZE,
            per_device_eval_batch_size=BATCH_SIZE,
            warmup_steps=500,
            weight_decay=0.01,
            learning_rate=LEARNING_RATE,
            logging_steps=200,
            eval_strategy='epoch',
            save_strategy='epoch',
            load_best_model_at_end=True,
            metric_for_best_model='accuracy',
            fp16=torch.cuda.is_available(),
            gradient_accumulation_steps=GRAD_ACCUM,
            report_to='none',
            seed=SEED
        )

    def evaluate_model(trainer, val_dataset, val_labels, label_names, model_name):
        predictions  = trainer.predict(val_dataset)
        pred_labels  = predictions.predictions.argmax(axis=-1)
        accuracy     = accuracy_score(val_labels, pred_labels)
        p, r, f1_w, _ = precision_recall_fscore_support(val_labels, pred_labels, average='weighted')
        _, _, f1_m, _ = precision_recall_fscore_support(val_labels, pred_labels, average='macro')
        print(f"\n--- {model_name} Results ---")
        print(f"Accuracy    : {accuracy:.4f}")
        print(f"Weighted F1 : {f1_w:.4f}")
        print(f"Macro F1    : {f1_m:.4f}")
        print("\nClassification Report:")
        print(classification_report(val_labels, pred_labels, target_names=label_names))
        return pred_labels, accuracy, f1_m, f1_w

    def clear_gpu():
        torch.cuda.empty_cache()
        gc.collect()

    def analyze_noise(series, name):
        texts = series.astype(str)
        total = len(texts)
        checks = {
            'URLs'          : texts.str.contains(r'http\S+|www\S+', regex=True).sum(),
            '@Mentions'     : texts.str.contains(r'@\w+', regex=True).sum(),
            '#Hashtags'     : texts.str.contains(r'#\w+', regex=True).sum(),
            'HTML tags'     : texts.str.contains(r'<.*?>', regex=True).sum(),
            'Numbers'       : texts.str.contains(r'\d+', regex=True).sum(),
            'Multi-punct'   : texts.str.contains(r'[!?]{2,}', regex=True).sum(),
            'All uppercase' : texts.apply(lambda x: x.isupper()).sum(),
            'Non-ASCII'     : texts.apply(lambda x: any(ord(c) > 127 for c in x)).sum(),
        }
        print(f"\nNoise Analysis - {name} (n={total:,}):")
        for label, count in checks.items():
            print(f"  {label:15}: {count:6}  ({count/total*100:.1f}%)")

    analyze_noise(df['text'], 'Emotion Train')
    analyze_noise(go_train_df['text'], 'GoEmotions Train')

    # Step 17: Text cleaning function (lowercase version - for uncased models)
    # Preserves emotional punctuation (!, ?) as they carry sentiment signals

    def clean_text(text):
        text = str(text)
        text = re.sub(r'http\S+|www\S+', '', text)     # remove URLs
        text = re.sub(r'@\w+', '', text)                # remove @mentions
        text = re.sub(r'#\w+', '', text)                # remove #hashtags
        text = re.sub(r'<.*?>', '', text)               # remove HTML tags
        text = text.encode('ascii', 'ignore').decode('ascii')  # remove non-ASCII
        text = re.sub(r'\s+', ' ', text).strip()        # collapse whitespace
        text = text.lower()                             # lowercase
        return text

    # Version for cased models (XLNet) - same but no lowercasing
    def clean_text_cased(text):
        text = str(text)
        text = re.sub(r'http\S+|www\S+', '', text)
        text = re.sub(r'@\w+', '', text)
        text = re.sub(r'#\w+', '', text)
        text = re.sub(r'<.*?>', '', text)
        text = text.encode('ascii', 'ignore').decode('ascii')
        text = re.sub(r'\s+', ' ', text).strip()
        return text

    def top_misclassifications(true_labels, pred_labels, label_names, model_name, top_n=5):
        cm    = confusion_matrix(true_labels, pred_labels)
        pairs = []
        for i in range(len(cm)):
            for j in range(len(cm)):

        pairs.sort(key=lambda x: x[2], reverse=True)

    def predict_emotion(text, model, tokenizer, label_names):
        model.eval()
        cleaned = clean_text(text)
        inputs  = tokenizer(cleaned, return_tensors='pt', truncation=True, max_length=MAX_LENGTH)
        if torch.cuda.is_available():
            inputs = {k: v.to(DEVICE) for k, v in inputs.items()}
            model.to(DEVICE)
        with torch.no_grad():
            logits     = model(**inputs).logits
            pred_id    = logits.argmax(dim=-1).item()
            confidence = torch.softmax(logits, dim=-1).squeeze()[pred_id].item()

    # Academic Document: Comparative Evaluation of Large Language Models in Emotion Detection

    ## Abstract
    This document presents a comparative analysis of five prominent Large Language Models (LLMs) – BERT, RoBERTa, DistilBERT, ALBERT, and XLNet – for emotion detection across two distinct datasets: a 6-class basic emotion dataset and a 28-class fine-grained GoEmotions dataset. The study aims to evaluate the performance of these models in varying degrees of emotional granularity and to elucidate the inherent challenges faced by LLMs in distinguishing subtle emotional nuances. While models achieved high accuracy on basic emotion classification (up to ~94-95%), a significant performance degradation was observed in the fine-grained task (accuracy ~58%), primarily attributed to semantic blurring, severe class imbalance, and the complex nature of informal textual data. This analysis provides a justified understanding of the limitations and capabilities of current LLMs in the nuanced field of affective computing.

    ## 1. Introduction
    Emotion detection in text is a critical task within Natural Language Processing (NLP) with applications ranging from sentiment analysis to mental health monitoring. However, the complexity of human emotions, particularly their fine-grained distinctions, poses a significant challenge for automated systems. This study investigates the performance disparities of several state-of-the-art Transformer-based LLMs when confronted with both broadly defined (6-class) and highly nuanced (28-class) emotional categories. The primary objective is to quantitatively assess this performance gap and qualitatively justify the observed limitations, thereby contributing to a deeper understanding of LLMs' capabilities in this domain.

    ## 2. Methodology

    ### 2.1 Models Evaluated
    Five popular LLM architectures were fine-tuned and evaluated:
    *   **BERT (bert-base-uncased)**: A foundational bidirectional Transformer model.
    *   **RoBERTa (roberta-base)**: An optimized version of BERT, trained with a larger dataset and longer training steps.
    *   **DistilBERT (distilbert-base-uncased)**: A smaller, faster, and lighter version of BERT, maintaining much of its performance.
    *   **ALBERT (albert-base-v2)**: A Lite BERT, utilizing parameter-reduction techniques to lower memory consumption and increase training speed.
    *   **XLNet (xlnet-base-cased)**: A generalized autoregressive pretraining method that combines the advantages of autoencoding and autoregressive models.

    ### 2.2 Datasets
    Two distinct datasets were utilized:
    *   **Emotion Dataset (6 classes)**: Comprising common emotions such as 'sadness', 'joy', 'love', 'anger', 'fear', and 'surprise'. This dataset served as a benchmark for basic emotion detection. (Train: 15,969, Validation: 1,998, Test: 2,000)
    *   **GoEmotions Dataset (28 classes)**: A fine-grained emotion dataset derived from Reddit comments, encompassing a broad spectrum of human emotions including 'admiration', 'amusement', 'anger', 'annoyance', 'approval', etc. For this study, a single-label classification approach was adopted by selecting the first reported label for multi-labeled instances. (Train: 43,188, Validation: 5,415, Test: 5,421)

    ### 2.3 Preprocessing
    Text data underwent cleaning, including removal of URLs, mentions, hashtags, HTML tags, non-ASCII characters, and normalization of whitespace. Lowercasing was applied for uncased models, while XLNet used a cased version. Duplicate and very short texts were removed. Tokenization was performed using the respective model's tokenizer with `max_length=128` and truncation.

    ### 2.4 Training and Evaluation
    Models were fine-tuned for 3 epochs (2 for GoEmotions BERT) with a batch size of 8 and a learning rate of 2e-5. Performance was evaluated using Accuracy, Macro F1-score, and Weighted F1-score on the validation sets. Confusion matrices and misclassification patterns were also analyzed.

    ## 3. Results: 6-Class Emotion Detection
    The models demonstrated strong performance on the 6-class Emotion dataset, indicating their capability in distinguishing broader emotional categories. The comparative results are presented below:

5. Discussion and Justification for Performance Discrepancy

The substantial performance difference between the 6-class and 28-class
emotion detection tasks can be attributed to several critical factors,
highlighting the inherent limitations of current LLMs in highly nuanced
emotional analysis:

1.  Complexity of Fine-Grained Emotions (28 vs. 6 Classes):
    Distinguishing between 28 subtle emotional states (e.g.,
    'admiration', 'approval', 'caring', 'optimism') is inherently more
    challenging than classifying into 6 broad categories. The increased
    number of labels necessitates a finer understanding of linguistic
    nuances, which LLMs often struggle to capture reliably.

2.  Severe Class Imbalance: The GoEmotions dataset exhibits extreme
    class imbalance. For instance, 'neutral' has 12,912 samples, while
    'pride' has only 57 and 'grief' has 65 samples in the training set,
    leading to an imbalance ratio of 226.53x. This causes models to
    heavily favor majority classes and perform poorly on minority
    classes, as evidenced by zero F1-scores for several rare emotions.

3.  Semantic Blurring/Overlapping Categories: Many fine-grained emotions
    share significant semantic overlap (e.g., 'sadness' and
    'disappointment', 'joy' and 'excitement', 'anger' and 'annoyance').
    The model frequently misclassifies between these closely related
    states because the textual cues are often ambiguous or shared,
    making precise differentiation difficult. The confusion matrices
    clearly show such misclassifications.

4.  Lack of Contextual Understanding: LLMs, despite their advancements,
    often struggle with deep contextual understanding, including
    implicit meanings, sarcasm, irony, and the speaker's true intent.
    This is particularly crucial in informal text like Reddit comments,
    where emotional expression is highly contextual and non-literal,
    leading to misinterpretations.

5.  Weak Emotional Display in Text: Fine-grained emotions might not
    always be expressed with strong, explicit linguistic markers in
    short text snippets. Subtle emotional signals are harder for models
    to pick up compared to more direct expressions associated with basic
    emotions, leading to weaker representations and poorer
    classification.

6.  Data Scarcity for Fine-Grained Distinctions: Even with a large total
    dataset, the effective sample size for many specific fine-grained
    emotions becomes very small due to the number of classes. This
    scarcity of unique examples for each subtle emotion hinders the
    model's ability to learn robust, distinct features for each class.
